# 1. Install ROS Noetic Desktop-Full
# Follow the guide here if not already installed:
# https://wiki.ros.org/noetic/Installation/Ubuntu

# 2. Install ultralytics (YOLO)
pip install ultralytics

# 3. Create catkin workspace
mkdir -p ~/catkin_ws/src
cd ~/catkin_ws/src

# 4. Copy or unzip your ROS package (detect_box) into src/
#    e.g. if you have detect_box.zip:
unzip /path/to/detect_box.zip -d ~/catkin_ws/src

# 5. Build your workspace
cd ~/catkin_ws
catkin_make

# 6. Set up environment variables
echo "source /opt/ros/noetic/setup.bash" >> ~/.bashrc
echo "source ~/catkin_ws/devel/setup.bash" >> ~/.bashrc
echo 'export GAZEBO_MODEL_PATH=~/catkin_ws/src/detect_box/model:$GAZEBO_MODEL_PATH' >> ~/.bashrc

source ~/.bashrc

# 7. To launch the package
roslaunch detect_box detect_box.launch

# 8. To run the camera script
rosrun detect_box camera.py
